```nagi title="counter_actor.nagi"
import std.actor as actor

class Context:
    initial: i64

async def create(context: shared[Context]) -> Result[i64, Error]:
    return ok(context.initial)

async def handle(state: i64, amount: i64) -> Result[actor.Turn[i64, i64, Error], Error]:
    next = state + amount
    return ok(actor.turn[i64, i64, Error](next, ok(next)))

async def monitor(group: actor.Supervisor[Context]) -> Result[unit, Error]:
    return await actor.run(group)

async def use_counter(counter: actor.Actor[i64, i64, Error], control: actor.Control) -> Result[unit, Error]:
    match await actor.ready(view(counter), 1000):
        case Ok(_):
            assert_true(True)
        case Err(_):
            return error("counter did not start")
    match await actor.call(view(counter), 5, 1000, 1000):
        case Ok(reply):
            match reply:
                case Ok(value):
                    print(value)
                case Err(_):
                    return error("counter rejected the message")
        case Err(_):
            return error("counter was unavailable")
    try await actor.shutdown(view(control))
    return ok(assert_true(True))

async def main() -> Result[unit, Error]:
    group = actor.supervisor[Context](Context(initial=0), actor.default_options())
    counter = try actor.register[i64, i64, i64, Error](view(group), "counter", create, handle, actor.default_actor_options())
    control = actor.control(view(group))
    async with scope:
        spawn monitor(group)
        spawn use_counter(counter, control)
    return ok(assert_true(True))
```

Only `create` constructs the initial state. `handle` returns the next state and reply from the current state and message. The two `1000` values in `actor.call` are enqueue and reply deadlines. Business `Err`, worker/call failure, and handler failure remain separate layers. After `shutdown`, the scope joins the monitor.

## Support and isolation

Actors and Supervisors are released for the Rust target in Nagi 0.2.3. They use Tokio tasks in one process and do not promise BEAM-style heap isolation, OOM isolation, or preemptive scheduling. Actors and Supervisors are unsupported on the Go target.
